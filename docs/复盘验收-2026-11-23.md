```sh
tmp/review-python/bin/python -Bu server/scripts/sync_market_review.py --provider baostock --refresh-date 2026-11-23
tmp/review-python/bin/python -Bm unittest discover -s server/scripts -p 'test_market_review*.py'
(cd server && REVIEW_AUDIT_DATE=2026-11-23 go test -count=1 -run '^TestPublishedMarketReview$' -v ./internal/service)
```
