# 开放云服务器数据库

```bash
// CentOS Linux 7 上配置任何 IP 都能够访问数据库
sudo vi /etc/my.cnf
// 在 [mysqld] 部分添加或修改以下行：这行配置会让 MySQL 监听所有网络接口。
bind-address = 0.0.0.0
// 重启 MySQL 服务： 保存并退出编辑器后，重启 MySQL 服务以应用更改。
sudo systemctl restart mysqld

// 这里的 username 和 password 需要替换为实际的用户名和密码。% 表示允许任何 IP 地址访问。
GRANT ALL PRIVILEGES ON *.* TO 'username'@'%' IDENTIFIED BY 'cc~@_1982_(?+?*Q-P=!!)';
(GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' IDENTIFIED BY 'cc~@_1982_(?+?*Q-P=!!)';)
FLUSH PRIVILEGES;
```



## 指定打包类型：

```go
// go build
GOOS=linux GOARCH=amd64 go build main.go
// linux start 测试权限
chmod +x main
// 执行部署（文件地址）
./main
// ！后台执行 nohup[命令及参数] [& 表示在后台运行]
sudo nohup ./main & // sudo nohup /www/wwwroot/9999110.xyz/server/main & 
// 查看后台相关运行情况
sudo netstat -tlnp
// 停止服务
sudo kill PID
```

# Nginx 配置

```nginx
# /api/ 代理地址，注意SSL证书
server {
   listen 80;
   listen 443 ssl;
   listen 443 quic;
  
  location /api/ {
        # 在 proxy_pass 后面添加 / ，表示将 /api/ 替换为空
        proxy_pass http://127.0.0.1:9981/; # 假设 Go 服务监听在本地 9981 端口
    		proxy_set_header Host $host;
    		proxy_set_header X-Real-IP $remote_addr;
    		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    		proxy_set_header X-Forwarded-Proto $scheme;

    		# 代理超时设置
    		proxy_connect_timeout 3000;
    		proxy_send_timeout 3000;
    		proxy_read_timeout 3000;

    	# 缓冲区设置
	    proxy_buffer_size 128k;
	    proxy_buffers 4 256k;
	    proxy_busy_buffers_size 256k;
	    proxy_temp_file_write_size 256k; # 修改为 256k 或更大
	}
}
```
